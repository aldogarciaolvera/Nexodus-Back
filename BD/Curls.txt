### Pantry (Despensa del Usuario) Endpoints ###

# 1. Obtener Despensa del Usuario
curl -X GET http://localhost:5079/api/pantry \
-H "Authorization: Bearer <TOKEN>"

# 2. Agregar Ingrediente a la Despensa
curl -X POST http://localhost:5079/api/pantry \
-H "Content-Type: application/json" \
-H "Authorization: Bearer <TOKEN>" \
-d '{
  "name": "Huevos",
  "quantity": 12,
  "unit": "unidades",
  "category": "Dairy",
  "expirationDate": "2026-11-30T00:00:00Z"
}'

# 3. Agregar Múltiples Ingredientes en Lote
curl -X POST http://localhost:5079/api/pantry/bulk \
-H "Content-Type: application/json" \
-H "Authorization: Bearer <TOKEN>" \
-d '[
  { "name": "Arroz integral", "quantity": 1000, "unit": "g", "category": "Carb" },
  { "name": "Aceite de oliva", "quantity": 500, "unit": "ml", "category": "Fat" },
  { "name": "Avena en hojuelas", "quantity": 800, "unit": "g", "category": "Carb" }
]'

# 4. Actualizar Ingrediente en Despensa
curl -X PUT http://localhost:5079/api/pantry/<PANTRY_ITEM_ID> \
-H "Content-Type: application/json" \
-H "Authorization: Bearer <TOKEN>" \
-d '{
  "quantity": 6,
  "unit": "unidades"
}'

# 5. Eliminar Ingrediente de Despensa
curl -X DELETE http://localhost:5079/api/pantry/<PANTRY_ITEM_ID> \
-H "Authorization: Bearer <TOKEN>"


### Habits & Habit Tracking Endpoints ###

# 1. Crear Nuevo Hábito
curl -X POST http://localhost:5079/api/habits \
-H "Content-Type: application/json" \
-H "Authorization: Bearer <TOKEN>" \
-d '{
  "name": "Beber 2.5L de Agua",
  "description": "Mantenerse hidratado durante el día laboral",
  "icon": "water_drop",
  "color": "#00E5FF",
  "category": "Health",
  "habitType": "Build",
  "trackingType": "Numeric",
  "targetValue": 2500,
  "targetUnit": "ml",
  "onTapAction": "ToggleOrAddOne",
  "frequencyType": "Daily",
  "streakUnit": "Days",
  "autoCompleteSource": "Nexodus.Water",
  "reminders": [
    { "time": "09:00", "daysOfWeek": "Monday,Tuesday,Wednesday,Thursday,Friday", "isEnabled": true },
    { "time": "15:00", "daysOfWeek": "Monday,Tuesday,Wednesday,Thursday,Friday", "isEnabled": true }
  ]
}'

# 2. Obtener Todos los Hábitos (con Cuadrícula Reciente)
curl -X GET "http://localhost:5079/api/habits?includeArchived=false&recentDays=60" \
-H "Authorization: Bearer <TOKEN>"

# 3. Obtener Detalle de un Hábito por ID
curl -X GET "http://localhost:5079/api/habits/<HABIT_ID>?recentDays=60" \
-H "Authorization: Bearer <TOKEN>"

# 4. Actualizar Configuración de un Hábito
curl -X PUT http://localhost:5079/api/habits/<HABIT_ID> \
-H "Content-Type: application/json" \
-H "Authorization: Bearer <TOKEN>" \
-d '{
  "name": "Beber 3L de Agua",
  "targetValue": 3000,
  "color": "#00B0FF"
}'

# 5. Realizar Check-in Rápido de un Hábito (Touch / Increment / Value)
curl -X POST http://localhost:5079/api/habits/<HABIT_ID>/check-in \
-H "Content-Type: application/json" \
-H "Authorization: Bearer <TOKEN>" \
-d '{
  "date": "2026-11-16",
  "action": "AddAmount",
  "amount": 500,
  "note": "Botella de agua después de entrenar"
}'

# 6. Obtener Historial de Celdas (Heatmap / Cuadrícula de Días)
curl -X GET "http://localhost:5079/api/habits/<HABIT_ID>/history?days=90" \
-H "Authorization: Bearer <TOKEN>"

# 7. Obtener Estadísticas Avanzadas del Hábito
curl -X GET http://localhost:5079/api/habits/<HABIT_ID>/stats \
-H "Authorization: Bearer <TOKEN>"

# 8. Reordenar Hábitos en la Pantalla Principal
curl -X POST http://localhost:5079/api/habits/reorder \
-H "Content-Type: application/json" \
-H "Authorization: Bearer <TOKEN>" \
-d '{
  "orderedHabitIds": [
    "<HABIT_ID_1>",
    "<HABIT_ID_2>",
    "<HABIT_ID_3>"
  ]
}'

# 9. Eliminar un Hábito
curl -X DELETE http://localhost:5079/api/habits/<HABIT_ID> \
-H "Authorization: Bearer <TOKEN>"
